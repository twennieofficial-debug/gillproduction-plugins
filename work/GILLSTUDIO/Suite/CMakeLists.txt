function(add_studio name code kind category)
 string(TOLOWER "${name}" lower)
 juce_add_plugin(${name} VERSION ${PROJECT_VERSION} COMPANY_NAME GILLPRODUCTION BUNDLE_ID "com.gillproduction.studio.${lower}" PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE ${code} PRODUCT_NAME ${name} FORMATS VST3 VST3_CATEGORIES Fx ${category} VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE)
 target_sources(${name} PRIVATE Source/Factory.cpp)
 target_compile_definitions(${name} PUBLIC GILL_STUDIO_KIND=${kind} JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(${name} PRIVATE StudioShared)
endfunction()
