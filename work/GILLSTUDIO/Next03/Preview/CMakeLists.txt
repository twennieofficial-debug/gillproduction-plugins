  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
 endif()
 if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "")

 if(TARGET juce_${m})
  set_property(TARGET juce_${m} PROPERTY INTERFACE_SOURCES "")
 endif()
endforeach()

set_target_properties(PreviewRuntime PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../GILLRESTORATION/build-msvc/GillRestorationRuntime.lib")
target_include_directories(PreviewRuntime INTERFACE "${JUCE_MODULES_DIR}")
target_compile_definitions(PreviewRuntime INTERFACE ${runtime_defs})

else()
 set(GILL_JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../dependencies/JUCE" CACHE PATH "Pinned JUCE source directory")
 set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
 set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 if(NOT TARGET juce::juceaide)
  add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
 endif()
 set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
  list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
 endforeach()
 add_library(PreviewRuntime STATIC Source/JuceRuntime.cpp)
 target_compile_definitions(PreviewRuntime PUBLIC ${runtime_defs})
 target_link_libraries(PreviewRuntime PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
endif()
add_library(PreviewShared STATIC Source/Processor.cpp Source/Editor.cpp)
target_link_libraries(PreviewShared PRIVATE PreviewRuntime)
target_include_directories(PreviewShared PUBLIC "${JUCE_MODULES_DIR}" "$<TARGET_PROPERTY:PreviewRuntime,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_definitions(PreviewShared PUBLIC ${runtime_defs} "$<TARGET_PROPERTY:PreviewRuntime,INTERFACE_COMPILE_DEFINITIONS>")
target_include_directories(PreviewShared PUBLIC Source ../../../GILLCommon)
juce_add_plugin(GILLSPECTRAPREVIEW VERSION ${PROJECT_VERSION} COMPANY_NAME GILLPRODUCTION BUNDLE_ID com.gillproduction.preview.spectra PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE Zsp1 PRODUCT_NAME "GILLSPECTRA PREVIEW" FORMATS VST3 VST3_CATEGORIES Fx Restoration VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE)
# The local JUCE export does not invalidate its generated Windows RC on a
# version-only reconfigure. Regenerate it once per version, never every build.
if(WIN32)
set(version_stamp "${CMAKE_CURRENT_BINARY_DIR}/preview-resource-version.txt")
set(previous_resource_version "")
if(EXISTS "${version_stamp}")
 file(READ "${version_stamp}" previous_resource_version)
endif()
if(NOT previous_resource_version STREQUAL PROJECT_VERSION)
 file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/GILLSPECTRAPREVIEW_artefacts/JuceLibraryCode/GILLSPECTRAPREVIEW_resources.rc")
 file(WRITE "${version_stamp}" "${PROJECT_VERSION}")
endif()
endif()
