cmake_minimum_required(VERSION 3.22)
project(GillQualityHost VERSION 0.9.0 LANGUAGES C CXX)
# Keep each Mach-O image's JUCE implementation private, including custom
# static runtimes and Objective-C++ units. VST3 entry points stay explicit exports.
if(APPLE)
    set(CMAKE_C_VISIBILITY_PRESET hidden)
    set(CMAKE_CXX_VISIBILITY_PRESET hidden)
    set(CMAKE_OBJC_VISIBILITY_PRESET hidden)
    set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
    set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
get_filename_component(WORK "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(GILL_JUCE_SOURCE_DIR "${WORK}/dependencies/JUCE" CACHE PATH "Pinned JUCE")
if(GILL_JUCE_EXPORT)
    include("${GILL_JUCE_EXPORT}")
else()
    add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
endif()
set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
    list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
endforeach()
add_executable(GillQualityHost ../QualityHost.cpp)
target_compile_definitions(GillQualityHost PRIVATE ${runtime_defs} JUCE_PLUGINHOST_VST3=1)
if(GILL_PREBUILT_RUNTIME)
    if(NOT WIN32)
        message(FATAL_ERROR "The prebuilt neutral runtime is Windows-only")
    endif()
    add_library(QualityHostRuntime STATIC IMPORTED)
    set_target_properties(QualityHostRuntime PROPERTIES IMPORTED_LOCATION "${GILL_PREBUILT_RUNTIME}")
    target_include_directories(QualityHostRuntime INTERFACE "${JUCE_MODULES_DIR}")
    target_link_libraries(QualityHostRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
    target_link_libraries(GillQualityHost PRIVATE QualityHostRuntime)
    target_include_directories(GillQualityHost PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
    target_sources(GillQualityHost PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors/juce_audio_processors.cpp" "${JUCE_MODULES_DIR}/juce_audio_processors_headless/juce_audio_processors_headless.cpp")
else()
    target_link_libraries(GillQualityHost PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
endif()
if(APPLE)
    enable_language(OBJCXX)
    target_sources(GillQualityHost PRIVATE ../MacQualityHost.mm)
    target_link_libraries(GillQualityHost PRIVATE "-framework AppKit")
endif()
