cmake_minimum_required(VERSION 3.22)
project(GillMasterDSPTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
find_package(Threads REQUIRED)
add_executable(GillMasterCoreTests CoreTests.cpp)
add_executable(GillMasterDeltaTests DeltaTests.cpp ../Source/DeltaEngine.cpp)
target_link_libraries(GillMasterDeltaTests PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(GillMasterDeltaTests PRIVATE advapi32 bcrypt)
endif()
enable_testing()
add_test(NAME MASTER_CORE COMMAND GillMasterCoreTests)
add_test(NAME MASTER_DELTA COMMAND GillMasterDeltaTests)

add_executable(GillMasterCeilingTests CeilingTests.cpp)
add_test(NAME MASTER_CEILING COMMAND GillMasterCeilingTests)
