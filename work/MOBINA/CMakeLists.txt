cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
 if(NOT DEFINED CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "")
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
 endif()
 if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
 endif()
endif()
project(Mobina VERSION 0.20.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
 set(CMAKE_C_VISIBILITY_PRESET hidden)
 set(CMAKE_CXX_VISIBILITY_PRESET hidden)
 set(CMAKE_OBJC_VISIBILITY_PRESET hidden)
 set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
 set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
 if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-ffp-contract=off)
 endif()
endif()
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(WIN32)
include("${CMAKE_CURRENT_SOURCE_DIR}/../GILLSTUDIO/Suite/JUCEExportConfig.cmake")
set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
 list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
 if(TARGET juce_${m})
  set_property(TARGET juce_${m} PROPERTY INTERFACE_SOURCES "")
 endif()
endforeach()
add_library(MobinaRuntime STATIC IMPORTED)
set_target_properties(MobinaRuntime PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../GILLRESTORATION/build-msvc/GillRestorationRuntime.lib")
target_include_directories(MobinaRuntime INTERFACE "${JUCE_MODULES_DIR}")
target_compile_definitions(MobinaRuntime INTERFACE ${runtime_defs})
target_link_libraries(MobinaRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
else()
 set(GILL_JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "Pinned JUCE source directory")
 set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
 set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 if(NOT TARGET juce::juceaide)
  add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
 endif()
 set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
  list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
 endforeach()
 add_library(MobinaRuntime STATIC Source/JuceRuntime.cpp)
 target_compile_definitions(MobinaRuntime PUBLIC ${runtime_defs})
 target_link_libraries(MobinaRuntime PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
endif()
add_library(MobinaShared STATIC Source/Processor.cpp Source/Editor.cpp)
target_link_libraries(MobinaShared PRIVATE MobinaRuntime)
target_include_directories(MobinaShared PUBLIC "${JUCE_MODULES_DIR}" "$<TARGET_PROPERTY:MobinaRuntime,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_definitions(MobinaShared PUBLIC ${runtime_defs} "$<TARGET_PROPERTY:MobinaRuntime,INTERFACE_COMPILE_DEFINITIONS>")
juce_add_plugin(MOBINA VERSION ${PROJECT_VERSION} COMPANY_NAME GILLPRODUCTION BUNDLE_ID com.gillproduction.mobina PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE Mbn1 PRODUCT_NAME MOBINA FORMATS VST3 VST3_CATEGORIES Fx Reverb Modulation VST3_AUTO_MANIFEST FALSE COPY_PLUGIN_AFTER_BUILD FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE)
target_sources(MOBINA PRIVATE Source/Factory.cpp)
target_compile_definitions(MOBINA PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MOBINA PRIVATE MobinaShared)
add_executable(MobinaTests Tests/Tests.cpp)
target_link_libraries(MobinaTests PRIVATE MobinaShared)
if(WIN32)
 add_executable(ThemeConcepts Tests/Concepts.cpp)
 target_link_libraries(ThemeConcepts PRIVATE MobinaRuntime)
endif()

enable_testing()
if(APPLE)
 enable_language(OBJCXX)
endif()
if(APPLE)
 target_sources(MobinaTests PRIVATE Tests/MacTestApplication.mm)
 target_link_libraries(MobinaTests PRIVATE "-framework AppKit")
endif()
add_test(NAME MOBINA_INTEGRATION COMMAND MobinaTests)
set_tests_properties(MOBINA_INTEGRATION PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 1800)
