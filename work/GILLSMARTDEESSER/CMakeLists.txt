cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT DEFINED CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "")
        set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
    endif()
    if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "")
        set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
    endif()
endif()
project(GILLSMARTDEESSER VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(-ffp-contract=off)
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
if(NOT WIN32 AND (GILL_JUCE_EXPORT OR GILL_PREBUILT_RUNTIME))
    message(FATAL_ERROR "Prebuilt JUCE runtime/export shortcuts are Windows-only")
endif()
set(GILL_JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "Pinned JUCE source directory")
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(GILL_JUCE_EXPORT)
    include("${GILL_JUCE_EXPORT}")
else()
    add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
endif()
set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
    list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
endforeach()
if(GILL_PREBUILT_RUNTIME)
    add_library(GillSmartDeEsserRuntime STATIC IMPORTED)
    set_target_properties(GillSmartDeEsserRuntime PROPERTIES IMPORTED_LOCATION "${GILL_PREBUILT_RUNTIME}")
    target_include_directories(GillSmartDeEsserRuntime INTERFACE "${JUCE_MODULES_DIR}")
    target_compile_definitions(GillSmartDeEsserRuntime INTERFACE ${runtime_defs})
    target_link_libraries(GillSmartDeEsserRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
    foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
        if(TARGET juce_${m})
            set_property(TARGET juce_${m} PROPERTY INTERFACE_SOURCES "")
        endif()
    endforeach()
else()
    add_library(GillSmartDeEsserRuntime STATIC Source/JuceRuntime.cpp)
    target_compile_definitions(GillSmartDeEsserRuntime PUBLIC ${runtime_defs})
    target_link_libraries(GillSmartDeEsserRuntime PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
    if(WIN32)
        target_link_libraries(GillSmartDeEsserRuntime PUBLIC advapi32)
    endif()
endif()
juce_add_binary_data(GillSmartDeEsserAssets HEADER_NAME BinaryData.h NAMESPACE BinaryData SOURCES Assets/core_reference.png)
add_library(GillSmartDeEsserShared STATIC Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(GillSmartDeEsserShared PRIVATE GillSmartDeEsserRuntime GillSmartDeEsserAssets)
target_include_directories(GillSmartDeEsserShared PUBLIC Source "$<TARGET_PROPERTY:GillSmartDeEsserRuntime,INTERFACE_INCLUDE_DIRECTORIES>" "${JUCE_MODULES_DIR}")
target_compile_definitions(GillSmartDeEsserShared PUBLIC ${runtime_defs} "$<TARGET_PROPERTY:GillSmartDeEsserRuntime,INTERFACE_COMPILE_DEFINITIONS>")
juce_add_plugin(GILLSMARTDEESSER VERSION ${PROJECT_VERSION} COMPANY_NAME "GILLPRODUCTION"
    BUNDLE_ID "com.gillproduction.gillsmartdeesser" PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE Gsd1
    PRODUCT_NAME "GILLSMARTDEESSER" DESCRIPTION "GILL Automatic Vocal De-Esser"
    FORMATS VST3 VST3_CATEGORIES Fx Dynamics VST3_AUTO_MANIFEST FALSE
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
if(WIN32)
    get_target_property(generated GILLSMARTDEESSER JUCE_GENERATED_SOURCES_DIRECTORY)
    get_target_property(info GILLSMARTDEESSER JUCE_INFO_FILE)
    add_custom_command(OUTPUT "${generated}/GILLSMARTDEESSER_resources.rc" APPEND DEPENDS "${info}")
endif()
target_sources(GILLSMARTDEESSER PRIVATE Source/Factory.cpp)
target_compile_definitions(GILLSMARTDEESSER PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(GILLSMARTDEESSER PRIVATE GillSmartDeEsserShared PUBLIC juce::juce_recommended_config_flags)

enable_testing()
add_executable(GillSmartDeEsserDspTests Tests/DspTests.cpp)
target_include_directories(GillSmartDeEsserDspTests PRIVATE Source)
add_test(NAME DSP COMMAND GillSmartDeEsserDspTests)
add_executable(GillSmartDeEsserLearningTests Tests/LearningTests.cpp)
target_include_directories(GillSmartDeEsserLearningTests PRIVATE Source)
add_test(NAME LEARNING COMMAND GillSmartDeEsserLearningTests)
add_executable(GillSmartDeEsserIntegrationTests Tests/PluginTests.cpp)
target_link_libraries(GillSmartDeEsserIntegrationTests PRIVATE GillSmartDeEsserShared)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(GillSmartDeEsserIntegrationTests PRIVATE Tests/MacTestApplication.mm)
    target_link_libraries(GillSmartDeEsserIntegrationTests PRIVATE "-framework AppKit")
endif()
add_test(NAME PLUGIN_INTEGRATION COMMAND GillSmartDeEsserIntegrationTests)
set_tests_properties(DSP LEARNING PLUGIN_INTEGRATION PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
