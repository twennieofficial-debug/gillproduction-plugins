function(add_gill product code kind category)
  string(TOLOWER "${product}" lower)
  juce_add_plugin(${product} VERSION ${PROJECT_VERSION} COMPANY_NAME "GILLPRODUCTION"
    BUNDLE_ID "com.gillproduction.${lower}" PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE ${code}
    PRODUCT_NAME "${product}" DESCRIPTION "${product} Vocal Effects"
    FORMATS VST3 VST3_CATEGORIES Fx ${category} VST3_AUTO_MANIFEST FALSE
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
  if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
    get_target_property(generated ${product} JUCE_GENERATED_SOURCES_DIRECTORY)
    get_target_property(info ${product} JUCE_INFO_FILE)
    add_custom_command(OUTPUT "${generated}/${product}_resources.rc" APPEND DEPENDS "${info}")
  endif()
  target_sources(${product} PRIVATE Source/Factory.cpp)
  target_compile_definitions(${product} PUBLIC GILL_DYNAMICS_KIND=${kind} JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(${product} PRIVATE GillDynamicsShared GillDynamicsAssets PUBLIC juce::juce_recommended_config_flags)
endfunction()
