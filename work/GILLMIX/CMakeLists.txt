cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT DEFINED CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "")
        set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
    endif()
    if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "")
        set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
    endif()
endif()
project(GILLMIX VERSION 0.10.0 LANGUAGES C CXX)
# Keep each Mach-O image's JUCE implementation private, including custom
# static runtimes and Objective-C++ units. VST3 entry points stay explicit exports.
if(APPLE)
    set(CMAKE_C_VISIBILITY_PRESET hidden)
    set(CMAKE_CXX_VISIBILITY_PRESET hidden)
    set(CMAKE_OBJC_VISIBILITY_PRESET hidden)
    set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
    set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(-ffp-contract=off)
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
if(NOT WIN32 AND (GILL_JUCE_EXPORT OR GILL_PREBUILT_RUNTIME))
    message(FATAL_ERROR "Prebuilt JUCE runtime/export shortcuts are Windows-only")
endif()
set(GILL_JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "Pinned JUCE source directory")
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(GILL_JUCE_EXPORT)
    include("${GILL_JUCE_EXPORT}")
else()
    add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
endif()
set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
    list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
endforeach()
if(GILL_PREBUILT_RUNTIME)
    add_library(GillMixRuntime STATIC IMPORTED)
    set_target_properties(GillMixRuntime PROPERTIES IMPORTED_LOCATION "${GILL_PREBUILT_RUNTIME}")
    target_include_directories(GillMixRuntime INTERFACE "${JUCE_MODULES_DIR}")
    target_compile_definitions(GillMixRuntime INTERFACE ${runtime_defs})
    target_link_libraries(GillMixRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
    foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
        if(TARGET juce_${m})
            set_property(TARGET juce_${m} PROPERTY INTERFACE_SOURCES "")
        endif()
    endforeach()
else()
    add_library(GillMixRuntime STATIC Source/JuceRuntime.cpp)
    target_compile_definitions(GillMixRuntime PUBLIC ${runtime_defs})
    target_link_libraries(GillMixRuntime PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
    if(WIN32)
        target_link_libraries(GillMixRuntime PUBLIC advapi32)
    endif()
endif()

juce_add_binary_data(GillMixAssets SOURCES Assets/core_reference.png)
add_library(GillMixShared STATIC Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(GillMixShared PRIVATE GillMixRuntime GillMixAssets)
target_include_directories(GillMixShared PUBLIC Source "$<TARGET_PROPERTY:GillMixRuntime,INTERFACE_INCLUDE_DIRECTORIES>" "${JUCE_MODULES_DIR}")
target_compile_definitions(GillMixShared PUBLIC ${runtime_defs} "$<TARGET_PROPERTY:GillMixRuntime,INTERFACE_COMPILE_DEFINITIONS>")
if(WIN32)
    target_link_libraries(GillMixShared PUBLIC bcrypt advapi32)
endif()
foreach(product GILLMIX GILLLINK)
    if(product STREQUAL "GILLMIX")
        set(code Gmx1)
        set(link_product 0)
    else()
        set(code Gln1)
        set(link_product 1)
    endif()
    string(TOLOWER "${product}" lower_name)
    juce_add_plugin(${product} VERSION ${PROJECT_VERSION} COMPANY_NAME "GILLPRODUCTION"
        BUNDLE_ID "com.gillproduction.${lower_name}" PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE ${code}
        PRODUCT_NAME "${product}" DESCRIPTION "GILL Explicit Session Level Assistant"
        FORMATS VST3 VST3_CATEGORIES Fx Tools VST3_AUTO_MANIFEST FALSE
        IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
    if(WIN32)
        get_target_property(generated ${product} JUCE_GENERATED_SOURCES_DIRECTORY)
        get_target_property(info ${product} JUCE_INFO_FILE)
        add_custom_command(OUTPUT "${generated}/${product}_resources.rc" APPEND DEPENDS "${info}")
    endif()
    target_sources(${product} PRIVATE Source/Factory.cpp)
    target_compile_definitions(${product} PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0 PRIVATE GILL_LINK_PRODUCT=${link_product})
    target_link_libraries(${product} PRIVATE GillMixShared GillMixAssets PUBLIC juce::juce_recommended_config_flags)
endforeach()
enable_testing()
add_executable(GillMixCoreTests Tests/CoreTests.cpp)
target_include_directories(GillMixCoreTests PRIVATE Source)
target_compile_definitions(GillMixCoreTests PRIVATE GILL_MIX_TESTING=1)
if(WIN32)
    target_link_libraries(GillMixCoreTests PRIVATE bcrypt advapi32)
endif()
add_test(NAME MIX_CORE COMMAND GillMixCoreTests)
add_executable(GillMixIntegrationTests Tests/PluginTests.cpp)
target_link_libraries(GillMixIntegrationTests PRIVATE GillMixShared GillMixAssets)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(GillMixIntegrationTests PRIVATE Tests/MacTestApplication.mm)
    target_link_libraries(GillMixIntegrationTests PRIVATE "-framework AppKit")
endif()
add_test(NAME MIX_INTEGRATION COMMAND GillMixIntegrationTests)
foreach(module A B)
    add_library(GillMixBusModule${module} SHARED Tests/BusModule.cpp)
    target_include_directories(GillMixBusModule${module} PRIVATE Source)
    if(WIN32)
        target_link_libraries(GillMixBusModule${module} PRIVATE bcrypt advapi32)
    endif()
endforeach()
add_executable(GillMixCrossDllTests Tests/CrossDllTests.cpp)
target_include_directories(GillMixCrossDllTests PRIVATE Source)
target_link_libraries(GillMixCrossDllTests PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(GillMixCrossDllTests PRIVATE bcrypt advapi32)
endif()
add_test(NAME MIX_CROSS_DLL COMMAND GillMixCrossDllTests "$<TARGET_FILE:GillMixBusModuleA>" "$<TARGET_FILE:GillMixBusModuleB>")
set_tests_properties(MIX_CORE MIX_INTEGRATION MIX_CROSS_DLL PROPERTIES TIMEOUT 180 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
