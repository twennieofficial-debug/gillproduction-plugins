cmake_minimum_required(VERSION 3.22)
project(Update22Regressions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_C_VISIBILITY_PRESET hidden)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/O2 /fp:precise /utf-8)
else()
  add_compile_options(-O2 -fno-fast-math -ffp-contract=off)
endif()
if(NOT SPECTRAL_ONLY)
  if(GILL_JUCE_EXPORT)
    include("${GILL_JUCE_EXPORT}")
    add_library(RegressionRuntime STATIC IMPORTED)
    set_target_properties(RegressionRuntime PROPERTIES IMPORTED_LOCATION "${GILL_PREBUILT_RUNTIME}")
    target_include_directories(RegressionRuntime INTERFACE "${JUCE_MODULES_DIR}")
    target_link_libraries(RegressionRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
  else()
    add_subdirectory("${SOURCE_ROOT}/dependencies/JUCE" JUCE)
    add_library(RegressionRuntime INTERFACE)
    target_link_libraries(RegressionRuntime INTERFACE juce::juce_audio_utils juce::juce_recommended_config_flags)
  endif()
  target_compile_definitions(RegressionRuntime INTERFACE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  foreach(mode Current Baseline)
    if(mode STREQUAL "Current")
      set(root "${SOURCE_ROOT}")
    else()
      set(root "${CMAKE_CURRENT_SOURCE_DIR}/baseline")
    endif()
    add_executable(Suite${mode} SuiteNative.cpp "${root}/GILLSTUDIO/Suite/Source/Processor.cpp")
    target_include_directories(Suite${mode} PRIVATE "${root}/GILLSTUDIO/Suite/Source")
    target_link_libraries(Suite${mode} PRIVATE RegressionRuntime)
  endforeach()
  add_executable(Update22Host Host.cpp)
  target_compile_definitions(Update22Host PRIVATE JUCE_PLUGINHOST_VST3=1)
  target_link_libraries(Update22Host PRIVATE RegressionRuntime)
  if(GILL_JUCE_EXPORT)
    target_include_directories(Update22Host PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
    target_sources(Update22Host PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors/juce_audio_processors.cpp" "${JUCE_MODULES_DIR}/juce_audio_processors_headless/juce_audio_processors_headless.cpp")
  endif()
endif()
foreach(case spectral-current spectral-baseline mono-current mono-baseline noise-controls level advanced frozen-regression)
  add_executable(${case} "${GENERATED}/${case}.cpp")
endforeach()
foreach(case Core Mechanisms)
  add_executable(Suite${case} "${SOURCE_ROOT}/GILLSTUDIO/Suite/Tests/${case}.cpp")
endforeach()
