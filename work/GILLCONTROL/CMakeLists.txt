cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT DEFINED CMAKE_OSX_ARCHITECTURES OR CMAKE_OSX_ARCHITECTURES STREQUAL "")
        set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
    endif()
    if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "")
        set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS")
    endif()
endif()
project(GILLCONTROL VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(-ffp-contract=off)
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()
if(NOT WIN32 AND (GILL_JUCE_EXPORT OR GILL_PREBUILT_RUNTIME))
    message(FATAL_ERROR "Prebuilt JUCE runtime/export shortcuts are Windows-only")
endif()
set(GILL_JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "Pinned JUCE source directory")
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
if(GILL_JUCE_EXPORT)
    include("${GILL_JUCE_EXPORT}")
else()
    add_subdirectory("${GILL_JUCE_SOURCE_DIR}" JUCE)
endif()
set(runtime_defs JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_WIN_WEBVIEW2=0 JUCE_REPORT_APP_USAGE=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
    list(APPEND runtime_defs JUCE_MODULE_AVAILABLE_juce_${m}=1)
endforeach()
if(GILL_PREBUILT_RUNTIME)
    add_library(GillControlRuntime STATIC IMPORTED)
    set_target_properties(GillControlRuntime PROPERTIES IMPORTED_LOCATION "${GILL_PREBUILT_RUNTIME}")
    target_include_directories(GillControlRuntime INTERFACE "${JUCE_MODULES_DIR}")
    target_compile_definitions(GillControlRuntime INTERFACE ${runtime_defs})
    target_link_libraries(GillControlRuntime INTERFACE juce::juce_recommended_config_flags winmm version shlwapi Imm32 comdlg32 ole32 oleaut32 uuid wininet ws2_32 advapi32)
    foreach(m audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_utils core data_structures events graphics gui_basics gui_extra)
        if(TARGET juce_${m})
            set_property(TARGET juce_${m} PROPERTY INTERFACE_SOURCES "")
        endif()
    endforeach()
else()
    add_library(GillControlRuntime STATIC Source/JuceRuntime.cpp)
    target_compile_definitions(GillControlRuntime PUBLIC ${runtime_defs})
    target_link_libraries(GillControlRuntime PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
    if(WIN32)
        target_link_libraries(GillControlRuntime PUBLIC advapi32)
    endif()
endif()
add_library(GillControlShared STATIC Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(GillControlShared PRIVATE GillControlRuntime)
target_include_directories(GillControlShared PUBLIC Source "$<TARGET_PROPERTY:GillControlRuntime,INTERFACE_INCLUDE_DIRECTORIES>" "${JUCE_MODULES_DIR}")
target_compile_definitions(GillControlShared PUBLIC ${runtime_defs} "$<TARGET_PROPERTY:GillControlRuntime,INTERFACE_COMPILE_DEFINITIONS>")
juce_add_plugin(GILLCONTROL VERSION ${PROJECT_VERSION} COMPANY_NAME "GILLPRODUCTION"
    BUNDLE_ID "com.gillproduction.gillcontrol" PLUGIN_MANUFACTURER_CODE Gill PLUGIN_CODE Gct6
    PRODUCT_NAME "GILLCONTROL" DESCRIPTION "GILL Session Quality Controller"
    FORMATS VST3 VST3_CATEGORIES Fx Tools VST3_AUTO_MANIFEST FALSE
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
if(WIN32)
    get_target_property(generated GILLCONTROL JUCE_GENERATED_SOURCES_DIRECTORY)
    get_target_property(info GILLCONTROL JUCE_INFO_FILE)
    add_custom_command(OUTPUT "${generated}/GILLCONTROL_resources.rc" APPEND DEPENDS "${info}")
endif()
target_sources(GILLCONTROL PRIVATE Source/Factory.cpp)
target_compile_definitions(GILLCONTROL PUBLIC JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(GILLCONTROL PRIVATE GillControlShared PUBLIC juce::juce_recommended_config_flags)

enable_testing()
foreach(module A B)
    add_library(GillQualityModule${module} SHARED Tests/BusModule.cpp)
    target_compile_definitions(GillQualityModule${module} PRIVATE GILL_QUALITY_NO_JUCE=1 GILL_QUALITY_TESTING=1)
    if(WIN32)
        target_link_libraries(GillQualityModule${module} PRIVATE advapi32)
    endif()
endforeach()
add_executable(GillQualityBusTests Tests/BusTests.cpp)
target_compile_definitions(GillQualityBusTests PRIVATE GILL_QUALITY_NO_JUCE=1 GILL_QUALITY_TESTING=1)
target_link_libraries(GillQualityBusTests PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(GillQualityBusTests PRIVATE advapi32)
endif()
add_test(NAME QUALITY_CROSS_DLL_IPC COMMAND GillQualityBusTests "$<TARGET_FILE:GillQualityModuleA>" "$<TARGET_FILE:GillQualityModuleB>")
add_executable(GillControlIntegrationTests Tests/PluginTests.cpp)
target_link_libraries(GillControlIntegrationTests PRIVATE GillControlShared)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(GillControlIntegrationTests PRIVATE Tests/MacTestApplication.mm)
    target_link_libraries(GillControlIntegrationTests PRIVATE "-framework AppKit")
endif()
add_test(NAME CONTROL_INTEGRATION COMMAND GillControlIntegrationTests)
set_tests_properties(QUALITY_CROSS_DLL_IPC CONTROL_INTEGRATION PROPERTIES TIMEOUT 120 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
